# clock8002 Buildroot boot config for Raspberry Pi 5

# Buildroot kernel image
kernel=Image

# Skip Pi 5 OS compatibility check (newer EEPROM bootloaders require this
# for non-RPiOS images whose DTB lacks the expected brcm,bcm2712 compatible)
os_check=0

# Run in 64-bit mode
arm_64bit=1

# Enable DRM VC4 V3D driver (Pi 5 specific overlay)
dtoverlay=vc4-kms-v3d-pi5
max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt
disable_fw_kms_setup=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

# Enable optional hardware interfaces
dtparam=i2c_arm=on
dtparam=audio=on

# GPIO/UART serial overlays (Pi 5 specific — BCM2711 uart2/uart3 overlays cause SError)
dtparam=uart0=on
dtoverlay=uart2-pi5
dtoverlay=uart3-pi5

[all]
# RTC battery charging at 3.0V
dtparam=rtc_bbat_vchg=3000000
