# clock8002 Buildroot boot config for Raspberry Pi 5
# Based on working piClock deployment + Buildroot requirements

# Buildroot kernel image
kernel=Image

# Enable optional hardware interfaces
dtparam=i2c_arm=on
dtparam=audio=on

# Enable DRM VC4 V3D driver
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt
disable_fw_kms_setup=1

# Run in 64-bit mode
arm_64bit=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

[cm4]
otg_mode=1

[cm5]
dtoverlay=dwc2,dr_mode=host

[all]
# RTC battery charging at 3.0V
dtparam=rtc_bbat_vchg=3000000

# GPIO/UART serial overlays
dtoverlay=uart1
dtoverlay=uart2
dtoverlay=uart3
dtparam=uart0=on
